export interface ColorDef {
  key: ColorKey;
  name: string;

export const COLORS: ColorDef[] = [
  { key: "paper", name: "Paper", hex: "#fff9c4", deepHex: "#e8d972" },
  { key: "blush", name: "Blush", hex: "#f8bbd0", deepHex: "#e991ae" },
  { key: "sky", name: "Sky", hex: "#b3e5fc", deepHex: "#7bc8ec" },
  { key: "sage", name: "Sage", hex: "#c8e6c9", deepHex: "#94cf98" },
  { key: "apricot", name: "Apricot", hex: "#ffe0b2", deepHex: "#f2bc73" },
  { key: "lavender", name: "Lavender", hex: "#d1c4e9", deepHex: "#a893d6" },
];

export function getColorHex(key: ColorKey): string {
  return COLORS.find((c) => c.key === key)?.hex ?? "#fff9c4";
}

export function getColorDeepHex(key: ColorKey): string {
  return COLORS.find((c) => c.key === key)?.deepHex ?? "#e8d972";
}

```tsx
import { createMemo, For, onCleanup } from "solid-js";
// Deep imports: the "lucide-solid" barrel pulls ~3500 icon modules in dev.
import Search from "lucide-solid/icons/search";
import Hash from "lucide-solid/icons/hash";
import Pin from "lucide-solid/icons/pin";
import Plus from "lucide-solid/icons/plus";
import Settings from "lucide-solid/icons/settings";
import { notes, getAllTags } from "../stores/notesStore";
import { debounce } from "../utils/debounce";

interface SidebarProps {
  searchQuery: string;
  onSearchChange: (q: string) => void;
  selectedTag: string;
  onTagSelect: (tag: string) => void;
  filterPinned: boolean;
  onFilterPinnedChange: (v: boolean) => void;
  onCreateNote: () => void;
  onOpenSettings: () => void;
}

export default function Sidebar(props: SidebarProps) {
  const tags = createMemo(() => getAllTags(notes));
  const pinnedCount = createMemo(() => notes.filter((n) => n.pinned).length);
  const debouncedSearch = debounce((q: string) => props.onSearchChange(q), 200);
  onCleanup(() => debouncedSearch.cancel());

  const itemClass = (active: boolean) =>
    `relative text-left text-xs px-2.5 py-1.5 rounded-sm transition-colors flex items-center gap-1.5 ${
      active
        ? "bg-neutral-100 font-medium text-ink"
        : "text-ink-muted hover:bg-neutral-50"
    }`;

  const ActiveBar = () => (
    <span class="absolute left-0 top-1 bottom-1 w-0.5 bg-ink rounded-full" />
  );

  return (
    <div class="w-56 bg-panel border-r border-line flex flex-col h-full shrink-0">
      {/* Search */}
      <div class="p-3">
        <div class="relative">
          <Search
            size={14}
            class="absolute left-2.5 top-1/2 -translate-y-1/2 text-ink-muted pointer-events-none"
          />
          <input
            type="text"
            placeholder="搜索便签..."
            value={props.searchQuery}
            onInput={(e) => debouncedSearch(e.currentTarget.value)}
            class="w-full pl-8 pr-3 py-1.5 text-sm border border-line rounded-sm bg-surface focus:outline-none focus:border-neutral-400 transition-colors"
          />
        </div>
      </div>

      {/* Tags */}
      <div class="flex-1 overflow-y-auto px-3">
        <div class="flex flex-col gap-0.5">
          <button
            onClick={() => props.onTagSelect("")}
            class={itemClass(props.selectedTag === "")}
          >
            {props.selectedTag === "" && <ActiveBar />}
            <span class="flex-1">全部</span>
            <span class="text-neutral-400">{notes.length}</span>
          </button>
          <button
            onClick={() => {
              // Capture before onTagSelect: MainView resets filterPinned to
              // false inside onTagSelect, which made this toggle stick on.
              const next = !props.filterPinned;
              props.onTagSelect("");
              props.onFilterPinnedChange(next);
            }}
            class={itemClass(props.filterPinned)}
          >
            {props.filterPinned && <ActiveBar />}
            <Pin size={12} />
            <span class="flex-1">已置顶</span>
            <span class="text-neutral-400">{pinnedCount()}</span>
          </button>
          <div class="my-1 border-t border-line" />
          <For each={tags()}>
            {(tag) => (
              <button
                onClick={() =>
                  props.onTagSelect(
                    props.selectedTag === tag.name ? "" : tag.name
                  )
                }
                class={itemClass(props.selectedTag === tag.name)}
              >
                {props.selectedTag === tag.name && <ActiveBar />}
                <Hash size={12} />
                <span class="flex-1 truncate">{tag.displayName}</span>
                <span class="text-neutral-400">{tag.count}</span>
              </button>
            )}
          </For>
        </div>
      </div>

      {/* New note button */}
      <div class="p-3 border-t border-line flex flex-col gap-1.5">
        <button
          onClick={props.onCreateNote}
          class="w-full py-2 text-sm bg-ink text-white rounded-sm hover:bg-neutral-800 active:scale-[0.98] transition flex items-center justify-center gap-1.5"
        >
          <Plus size={15} />
          新建便签
        </button>
        <button
          onClick={props.onOpenSettings}
          class="w-full py-1.5 text-xs text-ink-muted rounded-sm hover:bg-neutral-50 transition-colors flex items-center justify-center gap-1.5"
        >
          <Settings size={13} />
          设置
        </button>
      </div>
    </div>
  );
}
```

```tsx
import { createSignal, createEffect, onCleanup, Show } from "solid-js";
import type { Editor } from "@tiptap/core";
// Deep imports: the "lucide-solid" barrel pulls ~3500 icon modules in dev.
import Bold from "lucide-solid/icons/bold";
import Italic from "lucide-solid/icons/italic";
import UnderlineIcon from "lucide-solid/icons/underline";
import Strikethrough from "lucide-solid/icons/strikethrough";
import List from "lucide-solid/icons/list";
import ListOrdered from "lucide-solid/icons/list-ordered";
import ImageIcon from "lucide-solid/icons/image";
import { insertImageFromBlob } from "../lib/insertImageFromBlob";

interface NoteToolbarProps {
  editor: Editor | undefined | null;
  noteId: string;
  onImageError?: (msg: string) => void;
}

export default function NoteToolbar(props: NoteToolbarProps) {
  // TipTap state is not reactive for Solid; bump a version signal on every
  // editor transaction so isActive() re-evaluates.
  const [version, setVersion] = createSignal(0);
  createEffect(() => {
    const ed = props.editor;
    if (!ed) return;
    const bump = () => setVersion((v) => v + 1);
    ed.on("transaction", bump);
    onCleanup(() => ed.off("transaction", bump));
  });

  const isActive = (name: string) => {
    version();
    return props.editor?.isActive(name) ?? false;
  };

  const btnClass = (active: boolean) =>
    `p-1 rounded-sm transition-colors ${
      active ? "bg-black/10 text-ink" : "text-ink/50 hover:bg-black/5 hover:text-ink"
    }`;

  // Defensive: each onClick guards so a stale click event can't dereference
  // a torn-down editor.
  const run = (fn: (e: Editor) => void) => () => {
    if (!props.editor) return;
    fn(props.editor);
  };

  let fileInputRef: HTMLInputElement | undefined;

  const handleFileChange = (e: Event) => {
    const input = e.currentTarget as HTMLInputElement;
    const files = input.files;
    if (!files || !props.editor) return;
    const blobs = Array.from(files);
    // Allow re-selecting the same file later (otherwise the change event
    // would not fire when the user picks the same path twice).
    input.value = "";
    void (async () => {
      let failed = 0;
      for (const file of blobs) {
        if (!props.editor) return;
        const ok = await insertImageFromBlob(props.editor, file, props.noteId);
        if (!ok) failed++;
      }
      if (failed > 0) props.onImageError?.(`${failed} 张图片插入失败，请重试。`);
    })();
  };

  return (
    <Show when={props.editor}>
      <div class="flex gap-0.5 mb-2 flex-wrap items-center">
        <button
          class={btnClass(isActive("bold"))}
          onClick={run((e) => e.chain().focus().toggleBold().run())}
          title="加粗"
          aria-label="加粗"
        >
          <Bold size={15} />
        </button>
        <button
          class={btnClass(isActive("italic"))}
          onClick={run((e) => e.chain().focus().toggleItalic().run())}
          title="斜体"
          aria-label="斜体"
        >
          <Italic size={15} />
        </button>
        <button
          class={btnClass(isActive("underline"))}
          onClick={run((e) => e.chain().focus().toggleUnderline().run())}
          title="下划线"
          aria-label="下划线"
        >
          <UnderlineIcon size={15} />
        </button>
        <button
          class={btnClass(isActive("strike"))}
          onClick={run((e) => e.chain().focus().toggleStrike().run())}
          title="删除线"
          aria-label="删除线"
        >
          <Strikethrough size={15} />
        </button>
        <span class="w-px h-4 bg-black/10 mx-1" />
        <button
          class={btnClass(isActive("bulletList"))}
          onClick={run((e) => e.chain().focus().toggleBulletList().run())}
          title="无序列表"
          aria-label="无序列表"
        >
          <List size={15} />
        </button>
        <button
          class={btnClass(isActive("orderedList"))}
          onClick={run((e) => e.chain().focus().toggleOrderedList().run())}
          title="有序列表"
          aria-label="有序列表"
        >
          <ListOrdered size={15} />
        </button>
        <span class="w-px h-4 bg-black/10 mx-1" />
        <button
          class={btnClass(false)}
          onClick={() => fileInputRef?.click()}
          title="插入图片"
          aria-label="插入图片"
        >
          <ImageIcon size={15} />
        </button>
        <input
          ref={fileInputRef!}
          type="file"
          accept="image/*"
          class="hidden"
          onChange={handleFileChange}
        />
      </div>
    </Show>
  );
}
```

// Deep import: the "lucide-solid" barrel pulls ~3500 icon modules in dev.
import ChevronDown from "lucide-solid/icons/chevron-down";
import { type Settings, type ColorKey } from "../lib/types";
import ColorSwatchRow from "./ColorSwatchRow";

interface SettingsDialogProps {
  settings: Settings;
  onSave: (next: Settings) => void;
  onCancel: () => void;
}

const INTERVALS = [
  { value: 1000, label: "1 秒" },
  { value: 2000, label: "2 秒" },
  { value: 5000, label: "5 秒" },
  { value: 10000, label: "10 秒" },
];

function Toggle(props: { checked: boolean; onChange: (v: boolean) => void }) {
  return (
    <button
      type="button"
      role="switch"
      aria-checked={props.checked}
      onClick={() => props.onChange(!props.checked)}
      class="relative w-8 h-[18px] rounded-full transition-colors shrink-0"
      classList={{
        "bg-ink": props.checked,
        "bg-neutral-300": !props.checked,
      }}
    >
      <span
        class="absolute top-0.5 left-0.5 w-3.5 h-3.5 rounded-full bg-white transition-transform"
        classList={{ "translate-x-3.5": props.checked }}
      />
    </button>
  );
}

export default function SettingsDialog(props: SettingsDialogProps) {
  const [defaultColor, setDefaultColor] = createSignal<ColorKey>(
    props.settings.defaultColor
  );
  const [autoSaveIntervalMs, setAutoSaveIntervalMs] = createSignal(
    props.settings.autoSaveIntervalMs
  );
  const [trayOnClose, setTrayOnClose] = createSignal(props.settings.trayOnClose);

      launchOnStartup: launchOnStartup(),

  };

      class="fixed inset-0 z-50 flex items-center justify-center bg-black/25"
      onClick={props.onCancel}
